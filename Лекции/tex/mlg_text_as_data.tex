\begin{frame}{TF-IDF по формуле}
\color{ink}
  \begin{columns}[T,onlytextwidth]
    \column{0.40\textwidth}
      {\small
      \формула{w(t,d) = \mathrm{tf}(t,d)\cdot\mathrm{idf}(t)}
      \формула{\mathrm{idf}(t) = \ln\frac{1+N}{1+\mathrm{df}(t)} + 1}
      \vspace{0.2em}
      {\footnotesize
      $\mathrm{tf}$ — сколько раз слово $t$ стоит в сегменте $d$;\\
      $\mathrm{df}$ — в скольких сегментах оно есть;\\
      $N = 160$ — сегментов в корпусе.\par
      \vspace{0.4em}
      Затем вектор делят на его длину: так длинный и короткий сегменты
      становятся сравнимы.\par}}
    \column{0.56\textwidth}
      {\footnotesize\color{ash} s044: «Если значение не указано, используется
      значение по умолчанию.»\par}
      \vspace{0.2em}
      {\footnotesize\renewcommand{\arraystretch}{0.95}
      \begin{tabular}{@{}lrrrrr@{}}
        \toprule
        слово & tf & df & idf & tf$\cdot$idf & вес \\
        \midrule
        значение     & 2 & 1  & 5,39 & 10,78 & \числ{0,689} \\
        указано      & 1 & 1  & 5,39 & 5,39  & 0,345 \\
        используется & 1 & 1  & 5,39 & 5,39  & 0,345 \\
        умолчанию    & 1 & 2  & 4,98 & 4,98  & 0,319 \\
        если         & 1 & 3  & 4,70 & 4,70  & 0,300 \\
        по           & 1 & 11 & 3,60 & 3,60  & 0,230 \\
        не           & 1 & 16 & 3,25 & 3,25  & \кейс{0,208} \\
        \bottomrule
      \end{tabular}}\par
      \vspace{0.1em}
      {\footnotesize\color{ash} вес $=$ tf$\cdot$idf$\,/\,15{,}64$, где
      $15{,}64 = \sqrt{10{,}78^2 + 5{,}39^2 + \ldots + 3{,}25^2}$ — длина вектора\par}
  \end{columns}

  \врезка{\textbf{Вес TF-IDF не определяет переводческую важность:} «не» весит втрое меньше «значения», хотя потеря отрицания может изменить смысл.}
\end{frame}

      {\footnotesize\color{ash} A и B смотрят почти в одну сторону,
      C — в другую.}
    \column{0.66\textwidth}
      {\small
      \textbf{Скалярное произведение} — перемножить числа попарно и сложить.
      Для неотрицательных признаков общие большие координаты увеличивают результат:
      \формула{\mathbf a\cdot\mathbf b = a_1 b_1 + a_2 b_2 + \ldots + a_n b_n}
      \textbf{Косинус} — то же, делённое на длины векторов:
      \формула{\cos(\mathbf a,\mathbf b) = \frac{\mathbf a\cdot\mathbf b}{\|\mathbf a\|\,\|\mathbf b\|},
         \qquad \|\mathbf a\| = \sqrt{a_1^2 + \ldots + a_n^2}}
      Для TF-IDF он от 0 до 1: единица означает совпадение направлений, ноль — отсутствие общих ненулевых координат. Добавление слов может изменить направление.}
  \end{columns}

  \врезка{\textbf{Л.р. № 2, разделы 3–4:} на 13 запросах к памяти переводов

  \framesubtitle{Признаки — слова, значение — сколько раз слово встретилось}
  {\small
  $\mathbf A$ = «Не удалось подключиться к серверу»,\quad
  $\mathbf B$ = «…к базе данных»\par
  \vspace{0.4em}
  \centering
  \begin{tabular}{@{}l*{7}{c}@{}}
    \toprule
     & базе & данных & к & не & подключиться & серверу & удалось \\
    \midrule
    $\mathbf A$ & 0 & 0 & 1 & 1 & 1 & 1 & 1 \\
    $\mathbf B$ & 1 & 1 & 1 & 1 & 1 & 0 & 1 \\
    \bottomrule
  \end{tabular}\par
  \raggedright
  \формула{\mathbf A\cdot\mathbf B = 0\cdot1 + 0\cdot1 + 1\cdot1 + 1\cdot1 + 1\cdot1 + 1\cdot0 + 1\cdot1 = 4}
  \формула{\|\mathbf A\| = \sqrt{5},\quad \|\mathbf B\| = \sqrt{6},\qquad
     \cos(\mathbf A,\mathbf B) = \frac{4}{\sqrt5\cdot\sqrt6} = \frac{4}{\sqrt{30}} \approx 0{,}730}}

\begin{frame}{Модель: балл класса — взвешенная сумма n-грамм}
\color{ink}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      {\small
      \формула{z_k = \sum_{j=1}^{4662} w_{kj}\,x_j + b_k}
      $x_j$ — TF-IDF n-граммы $j$ в сегменте; у большинства — ноль.\par
      $w_{kj}$ — насколько n-грамма $j$ говорит «за» класс $k$: плюс — за,
      минус — против.\par
      $b_k$ — сдвиг (свободный член): балл класса, когда признаков нет.\par
      \vspace{0.3em}
      Это скалярное произведение: вектор сегмента на вектор весов
      класса.}
    \column{0.44\textwidth}
      {\color{navy}\bfseries Сколько тут чисел}\\[0.2em]
      {\small У каждого из 4 классов свой вес для каждой из 4662 n-грамм:
      $4 \times 4662 = 18\,648$ весов и 4 сдвига. До обучения все они —
      нули.\par
      \vspace{0.5em}
      Ответ модели — класс с наибольшим баллом. Учить её удобнее через
      вероятности — следующий слайд.}
  \end{columns}

  \врезка{\textbf{Пример.} У отложенного сегмента «Компания оставляет за собой
  право…» 192 n-граммы. Их вклады в балл юридического класса в сумме дают
  5,97, сдвиг — $-1{,}62$, итого $z = 4{,}35$.}
\end{frame}

\begin{frame}{Softmax: от баллов к вероятностям}
\color{ink}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      {\small
      \формула{p_k = \frac{e^{z_k}}{e^{z_1} + e^{z_2} + e^{z_3} + e^{z_4}}}
      Экспонента даёт положительные значения. Деление на их сумму нормирует распределение: вероятности в сумме равны 1. Порядок не меняется: больший балл — большая вероятность.\par
      \vspace{0.4em}
      {\footnotesize\color{ash} Отложенный сегмент после обучения: «Компания
      оставляет за собой право в любое время вносить изменения в настоящие
      Условия.»\par}}
    \column{0.49\textwidth}
      {\small
      \begin{tabular}{@{}lrrr@{}}
        \toprule
        класс & $z_k$ & $e^{z_k}$ & $p_k$ \\
        \midrule
        интерфейс    & $-1{,}62$ & 0,20  & 0,003 \\
        документация & $-0{,}69$ & 0,50  & 0,006 \\
        маркетинг    & $-2{,}04$ & 0,13  & 0,002 \\
        юридический  & $4{,}35$  & 77,45 & \числ{0,989} \\
        \midrule
        сумма        &           & 78,28 & 1 \\
        \bottomrule
      \end{tabular}}\par
      \vspace{0.3em}
      {\footnotesize $p_{\text{юр}} = 77{,}45 / 78{,}28 = 0{,}989$}
  \end{columns}

  \врезка{\textbf{Зачем именно экспонента.} Она усиливает разрывы: баллы юридического класса и документации различаются
  на 5,04, а вероятности — в
  $e^{5{,}04} \approx 155$ раз.}
\end{frame}

\begin{frame}{Градиент и шаг оптимизации}
\color{ink}
{\small Градиент — вектор частных производных функции потерь по параметрам. Он указывает направление наибольшего локального роста потерь.\par\vspace{0.6em}\[\displaystyle \mathbf w\leftarrow\mathbf w-\eta\nabla L(\mathbf w)\]\par\vspace{0.6em}$\eta$ — скорость обучения. Знак минус задаёт движение против градиента. Большой шаг может вызвать колебания или расходимость; слишком маленький замедляет обучение.}\врезка{SGD, Adam и RMSprop используют разные правила обновления параметров. Ни один выбор скорости не гарантирует одинакового результата для всех моделей.}
\end{frame}

\begin{frame}{Градиент нашей модели: ошибка $\times$ признак}
\color{ink}
  {\small
  Для логистической регрессии наклон по каждому весу считается одной
  формулой:
  \формула{\frac{\partial L}{\partial w_{kj}} \;=\; \frac{1}{N}\sum_{i=1}^{N}
     \underbrace{\bigl(p_{ik} - [\,y_i = k\,]\bigr)}_{\text{ошибка в вероятности}}
     \cdot \underbrace{x_{ij}}_{\text{признак}}}
  $[y_i = k]$ — единица, если $k$ — правильный класс $i$-го сегмента, иначе
  ноль. У правильного класса ошибка отрицательна (вероятность занижена), у
  остальных — положительна (завышена).}

  \vspace{0.3em}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      {\color{navy}\bfseries Что из формулы следует}\\[0.2em]
      {\small Вес n-граммы меняют только сегменты, где она есть: при $x_{ij}
      = 0$ слагаемое — ноль. Вес растёт у класса, которому модель недодаёт
      вероятность, и падает у остальных.}
    \column{0.48\textwidth}
      {\color{brick}\bfseries Первый шаг на нашем корпусе}\\[0.2em]
      {\small Все $p = \tfrac14$. У каждой строки интерфейса с «ить\пр»
      («Удалить…», «Отменить…») ошибка $\tfrac14 - 1 = -\tfrac34$ у
      интерфейса и $+\tfrac14$ у остальных классов. После шага вес «ить\пр»
      для интерфейса — $+0{,}048$, для юридического — $-0{,}019$.}
  \end{columns}
\end{frame}

\begin{frame}{N-граммная модель в формулах}
\color{ink}
  {\small
  Вероятность текста раскладывается в цепочку: каждый символ — при условии
  всех предыдущих.
  \формула{P(c_1 c_2 \ldots c_T) = P(c_1)\cdot P(c_2 \mid c_1)\cdot P(c_3 \mid c_1 c_2)
     \cdot\ldots\cdot P(c_T \mid c_1 \ldots c_{T-1})}
  В интерактивной модели параметр $n$ задаёт число предыдущих символов, а вероятность
  оценивает частотами в обучающем тексте:
  \формула{P(c_t \mid c_{t-n} \ldots c_{t-1}) \;\approx\;
     \frac{\#(c_{t-n} \ldots c_{t-1}\,c_t)}{\#(c_{t-n} \ldots c_{t-1})}}
  $\#(\ldots)$ — сколько раз цепочка встретилась. Обучение здесь — подсчёт,
  градиента нет.}

  \vspace{0.2em}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      {\footnotesize\textbf{Пример, $n = 3$.} В обучающем тексте после «сох»
      всегда шло «р»: «сохранены», «сохраняют», «сохранения». Модель ставит
      «р» с вероятностью \числ{0,921} после сглаживания.}
    \column{0.48\textwidth}
      {\footnotesize\textbf{Сглаживание.} Цепочка, которой не было, получила
      бы вероятность ноль. Поэтому оценку смешивают с более короткими
      контекстами; на интерактивной странице длинный контекст
      берётся с весом 0,7.}
  \end{columns}
\end{frame}

\begin{frame}{Дальние зависимости и внимание}
\color{ink}
{\small В рекуррентной сети информация передаётся через последовательность скрытых состояний. При самовнимании представление позиции напрямую зависит от других доступных позиций.\par\vspace{0.6em}Это облегчает использование дальнего контекста, но не гарантирует правильного согласования или разрешения анафоры.}\врезка{В авторегрессионном декодере маска закрывает будущие позиции. В двунаправленном кодировщике доступны обе стороны контекста.}
\end{frame}

\begin{frame}{Внимание: оценки связей и взвешенная сумма}
\color{ink}
\begin{columns}[T,onlytextwidth]
\column{.51\textwidth}{\small
Для каждой позиции вычисляются запрос $\mathbf q_i$, ключ $\mathbf k_i$ и значение $\mathbf v_i$.
\формула{a_{ij}=\mathrm{softmax}_j\!\left(\frac{\mathbf q_i\cdot\mathbf k_j}{\sqrt{d_k}}\right)}
Представление позиции $i$ — взвешенная сумма значений:
\формула{\mathbf h_i=\sum_j a_{ij}\mathbf v_j}}
\column{.45\textwidth}{\small
Скалярное произведение даёт оценку связи запроса и ключа. Softmax нормирует оценки по доступным позициям $j$.
\par\vspace{.5em}
$d_k$ — размерность ключа. Деление на $\sqrt{d_k}$ ограничивает масштаб оценок.
\par\vspace{.5em}
Запросы, ключи и значения — проекции входных векторов. Матрицы проекций подбираются при обучении.}
\end{columns}
\врезка{Полное внимание использует матрицу оценок размера $n\times n$. Маска может закрывать часть позиций; в авторегрессионной модели закрыто будущее.}
\end{frame}

\begin{frame}{BLEU и словесные совпадения}
\color{ink}
  \relax

\end{frame}

\begin{frame}{BLEU по формуле}
\color{ink}
  \framesubtitle{BLEU одного сегмента в том виде, в каком его считает Л.р. № 3}
  \begin{columns}[T,onlytextwidth]
    \column{0.44\textwidth}
      {\small
      \формула{p_n = \frac{m_n + 1}{t_n + 1}, \qquad n = 1, 2, 3, 4}
      \формула{\mathrm{BP} = \begin{cases} 1, & c > r,\\ e^{\,1 - r/c}, & c \le r \end{cases}}
      \формула{\mathrm{BLEU} = \mathrm{BP}\cdot\sqrt[k]{\,p_1\, p_2 \cdots p_k}}}
    \column{0.52\textwidth}
      {\small
      $t_n$ — сколько n-грамм слов в гипотезе; $m_n$ — сколько из них есть в
      эталоне (каждая — не больше раз, чем там встречается).\par
      \vspace{0.3em}
      $c$ и $r$ — длина гипотезы и эталона в словах. BP (brevity penalty) — штраф только за слишком
      короткий перевод.\par
      \vspace{0.3em}
      $k$ — сколько долей удалось посчитать: в гипотезе из двух слов нет ни
      3-грамм, ни 4-грамм.}
  \end{columns}

  \врезка{\textbf{+1 в числителе и знаменателе — сглаживание.} Без него одна
  нулевая доля обнуляет весь BLEU: произведение с нулём — ноль. Слова
  сравниваются в нижнем регистре, пунктуация отбрасывается.}
\end{frame}

\begin{frame}{Согласие разметчиков}
\color{ink}
  \relax
  {\small Разметка — это измерение, и у измерения есть надёжность. Если два
  человека размечают один материал по-разному, ненадёжна и разметка, и всё,
  что из неё выведено.}

  \vspace{0.5em}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      {\small
      \textbf{Каппа Коэна} — согласие с поправкой на ожидаемые случайные совпадения:
      \формула{\kappa = \frac{p_{\text{совп}} - p_{\text{случ}}}{1 - p_{\text{случ}}}}
      Пример Л.р. № 3: совпали 8 ответов из 10, случайно совпало бы 22\,\%:
      $\kappa = \tfrac{0{,}80 - 0{,}22}{1 - 0{,}22} \approx 0{,}744$ —
      значение для учебного примера.}
    \column{0.44\textwidth}
      \врезка{\textbf{Зачем вычитать случайные совпадения.} Если 80\,\% сегментов без
      ошибок, два разметчика, независимо назначающих метки с такими частотами, совпадут в
      \числ{65\,\%} случаев — просто потому, что оба чаще ставят «нет».
      Каппа у них — ноль.}
  \end{columns}

  \vspace{0.5em}
  {\small Расхождения примера — s003: точность или терминология? s006:
  терминология или стиль? Оба ответа можно обосновать, пока руководство для
  разметчиков не решило иначе. Написать его — профессиональная задача
  (Л.р. № 3, задание 8).}
\end{frame}

\color{ink}

Критерий оценки показывает только выбранную сторону качества.\par\vspace{0.6em}Лингвистический анализ помогает определить категории, подготовить инструкцию по разметке и объяснить ошибки по контексту.}\врезка{Для каждой задачи нужно согласовать, что считается правильным ответом, как его измеряют и какие ошибки остаются вне автоматической проверки.}
