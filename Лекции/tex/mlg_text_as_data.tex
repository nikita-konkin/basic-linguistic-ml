\begin{frame}{TF-IDF по формуле}
  \begin{columns}[T,onlytextwidth]
    \column{0.40\textwidth}
      {\small
      \формула{w(t,d) = \mathrm{tf}(t,d)\cdot\mathrm{idf}(t)}
      \формула{\mathrm{idf}(t) = \ln\frac{1+N}{1+\mathrm{df}(t)} + 1}
      \vspace{0.2em}
      {\footnotesize
      $\mathrm{tf}$ — сколько раз слово $t$ стоит в сегменте $d$;\\
      $\mathrm{df}$ — в скольких сегментах оно есть;\\
      $N = 160$ — сегментов в корпусе.\par
      \vspace{0.4em}
      Затем вектор делят на его длину: так длинный и короткий сегменты
      становятся сравнимы.\par}}
    \column{0.56\textwidth}
      {\footnotesize\color{ash} s044: «Если значение не указано, используется
      значение по умолчанию.»\par}
      \vspace{0.2em}
      {\footnotesize\renewcommand{\arraystretch}{0.95}
      \begin{tabular}{@{}lrrrrr@{}}
        \toprule
        слово & tf & df & idf & tf$\cdot$idf & вес \\
        \midrule
        значение     & 2 & 1  & 5,39 & 10,78 & \числ{0,689} \\
        указано      & 1 & 1  & 5,39 & 5,39  & 0,345 \\
        используется & 1 & 1  & 5,39 & 5,39  & 0,345 \\
        умолчанию    & 1 & 2  & 4,98 & 4,98  & 0,319 \\
        если         & 1 & 3  & 4,70 & 4,70  & 0,300 \\
        по           & 1 & 11 & 3,60 & 3,60  & 0,230 \\
        не           & 1 & 16 & 3,25 & 3,25  & \кейс{0,208} \\
        \bottomrule
      \end{tabular}}\par
      \vspace{0.1em}
      {\footnotesize\color{ash} вес $=$ tf$\cdot$idf$\,/\,15{,}64$, где
      $15{,}64 = \sqrt{10{,}78^2 + 5{,}39^2 + \ldots + 3{,}25^2}$ — длина вектора\par}
  \end{columns}

  \врезка{\textbf{TF-IDF измеряет редкость, а не важность:} «не» весит втрое
  меньше «значения», хотя для перевода отрицание важнее всего.}
\end{frame}

\begin{frame}{Почему векторы почти пустые: закон Ципфа}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      \includegraphics{fig/tad_zipf.pdf}
    \column{0.44\textwidth}
      {\small
      Частота слова примерно обратно пропорциональна его месту $r$ в
      частотном списке:
      \формула{f(r) \approx \frac{C}{r^{\,a}}}
      В логарифмическом масштабе — прямая с наклоном $-a$. В больших
      корпусах $a \approx 1$, на наших 947 словах — 0,55.\par
      \vspace{0.3em}
      Главное — хвост: \числ{583} словоформы из 689 (85\,\%) встретились
      по разу.}
  \end{columns}

  \врезка{\textbf{Для модели это значит:} почти каждое слово нового сегмента
  она видит впервые. Отсюда пустые векторы на следующем слайде и выигрыш n-грамм: часть слова повторяется, даже когда само слово — нет.}
\end{frame}

\begin{frame}{Шаг четвёртый: а нужно ли слово вообще}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      {\small
      \textbf{Символьные n-граммы:} признак — подстрока из нескольких
      букв. Понятия «слово» в модели нет.\par
      \vspace{0.4em}
      «сохранить» $\rightarrow$ \texttt{сох}, \texttt{охр}, \texttt{хра},
      \texttt{ран}, \texttt{ани}, \texttt{нит}, \texttt{ить}\par
      «сохранение» даёт те же \texttt{сох}, \texttt{охр}, \texttt{хра},
      \texttt{ран} — словоформы похожи без словаря и без правил.}
    \column{0.44\textwidth}
      {\small Тот же наивный Байес:\par
      \vspace{0.3em}
      \begin{tabular}{@{}lr@{}}
        слова               & 0,444 \\
        стемминг            & 0,538 \\
        символьные 2–4      & 0,612 \\
        символьные 3–5      & \числ{0{,}631} \\
      \end{tabular}\par
      \vspace{0.4em}
      Цена: 5418–7821 признаков вместо 679, и прочесть их человек не
      может.}
  \end{columns}

  \врезка{\textbf{Почему слова проигрывают на коротких строках.} У 44 из 160
  сегментов, когда они попадают в контрольную часть, нет ни одного слова,
  встречавшегося при обучении: вектор пустой. Пустых векторов из n-грамм нет ни одного.}

  \playground{https://konkin-nikita.ru/gd_learn/vec/}{векторизация текста}%

\end{frame}

\begin{frame}{Вектор — это просто список чисел}
  \begin{columns}[T,onlytextwidth]
    \column{0.44\textwidth}
      {\small
      Вектор длины $n$ — упорядоченный список из $n$ чисел:
      $\mathbf x = (x_1, \ldots, x_n)$. И всё.\par
      \vspace{0.3em}
      После векторизации сегмент — точка в пространстве, где измерений
      столько, сколько признаков. Представить 5418 измерений нельзя, и не
      нужно: всё, что мы делаем с векторами, определено для любого их
      числа.\par
      \vspace{0.4em}
      {\color{navy}\bfseries Похожие тексты — близкие точки.}}
    \column{0.52\textwidth}
      \includegraphics[width=\textwidth]{fig/tad_sparsity.pdf}\par
      {\footnotesize\color{ash} Корпус глазами модели, фрагмент: 16 сегментов
      $\times$ 48 случайных n-грамм. Закрашено 14 клеток из 768.\par}
  \end{columns}

  \врезка{\textbf{Корпус Л.р. № 1} после символьных n-грамм 2–4 —
  \числ{160} точек в \числ{5418}-мерном пространстве. Ненулевых координат
  \числ{2,1\,\%}: в среднем 115 на сегмент, у коротких строк интерфейса — 63.}

      \формула{\mathbf a\cdot\mathbf b = a_1 b_1 + a_2 b_2 + \ldots + a_n b_n}
      \textbf{Косинус} — то же, делённое на длины векторов:
      \формула{\cos(\mathbf a,\mathbf b) = \frac{\mathbf a\cdot\mathbf b}{\|\mathbf a\|\,\|\mathbf b\|},
         \qquad \|\mathbf a\| = \sqrt{a_1^2 + \ldots + a_n^2}}
      Для TF-IDF он от 0 до 1: единица — одинаковый состав, ноль — ни одного
      общего признака. Длина текста не влияет — в отличие от Левенштейна.}
  \end{columns}

  \врезка{\textbf{Л.р. № 2, разделы 3–4:} на 13 запросах к памяти переводов
  косинус и Левенштейн расходятся в 4 случаях, и в двух из них можно обосновать оба
  ответа.}

  \playground{https://konkin-nikita.ru/gd_learn/tm/}{память переводов}%
  {четыре меры похожести на одном запросе.}
\end{frame}

\begin{frame}{Что значит «обучить модель»}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      {\small
      У модели есть \textbf{веса} — числа, которые она может менять.\par
      \vspace{0.4em}
      Есть \textbf{функция потерь} — число, показывающее, насколько
      предсказания расходятся с правильными ответами.\par
      \vspace{0.4em}

      \числ{1{,}000}, на отложенных — \числ{0{,}744}. Первое число
      измеряет память, второе — способность обобщать.}
  \end{columns}

  \vspace{0.8em}
  {\small Модель воспроизводит то, что было в обучающих данных, — вместе

\end{frame}

\begin{frame}{Модель: балл класса — взвешенная сумма n-грамм}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      {\small
      \формула{z_k = \sum_{j=1}^{4662} w_{kj}\,x_j + b_k}
      $x_j$ — TF-IDF n-граммы $j$ в сегменте; у большинства — ноль.\par
      $w_{kj}$ — насколько n-грамма $j$ говорит «за» класс $k$: плюс — за,
      минус — против.\par
      $b_k$ — сдвиг (свободный член): балл класса, когда признаков нет.\par
      \vspace{0.3em}
      Это скалярное произведение: вектор сегмента на вектор весов
      класса.}
    \column{0.44\textwidth}
      {\color{navy}\bfseries Сколько тут чисел}\\[0.2em]
      {\small У каждого из 4 классов свой вес для каждой из 4662 n-грамм:
      $4 \times 4662 = 18\,648$ весов и 4 сдвига. До обучения все они —
      нули.\par
      \vspace{0.5em}
      Ответ модели — класс с наибольшим баллом. Учить её удобнее через
      вероятности — следующий слайд.}
  \end{columns}

  \врезка{\textbf{Пример.} У отложенного сегмента «Компания оставляет за собой
  право…» 192 n-граммы. Их вклады в балл юридического класса в сумме дают
  5,97, сдвиг — $-1{,}62$, итого $z = 4{,}35$.}
\end{frame}

\begin{frame}{Softmax: от баллов к вероятностям}

\begin{frame}{Градиент: как именно подбирают веса}
  {\small Перебрать все комбинации весов невозможно. \textbf{Градиент}
  отвечает на вопрос: если чуть-чуть изменить каждый вес, в какую сторону
  потери убывают быстрее всего? Дальше — шаг в эту сторону, пересчёт,
  снова шаг. Это градиентный спуск: спуск по склону в тумане, на ощупь.}
  \формула{\mathbf w \;\leftarrow\; \mathbf w - \eta\,\nabla L(\mathbf w)}
  {\footnotesize\color{ash} $\nabla L$ — для каждого из 18\,648 весов число
  «насколько вырастут потери, если этот вес чуть увеличить»; $\eta$ (эта) —
  размер шага; минус — потому что идём вниз.\par}

  \vspace{0.4em}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      {\color{navy}\bfseries Размер шага}\\[0.2em]
      {\small Слишком велик — модель перескакивает низину, потери скачут
      вверх. Слишком мал — обучение не заканчивается. SGD, Adam, RMSprop — разные правила того, как делать шаг.}
    \column{0.47\textwidth}
      {\color{brick}\bfseries Что из этого следует}\\[0.2em]
      {\small Обучение — пошаговый подбор: результат зависит от начальной
      точки и от случайности. Выводить формулу градиента не нужно; читать её
      полезно — это следующие слайды.}
  \end{columns}

  \playground{https://konkin-nikita.ru/gd_learn/}{градиентный спуск}%
  {шаги видны на поверхности потерь, и сразу заметно, когда спуск расходится.}
\end{frame}

\begin{frame}{Градиент нашей модели: ошибка $\times$ признак}
  {\small
  Для логистической регрессии наклон по каждому весу считается одной
  формулой:
  \формула{\frac{\partial L}{\partial w_{kj}} \;=\; \frac{1}{N}\sum_{i=1}^{N}
     \underbrace{\bigl(p_{ik} - [\,y_i = k\,]\bigr)}_{\text{ошибка в вероятности}}
     \cdot \underbrace{x_{ij}}_{\text{признак}}}
  $[y_i = k]$ — единица, если $k$ — правильный класс $i$-го сегмента, иначе
  ноль. У правильного класса ошибка отрицательна (вероятность занижена), у
  остальных — положительна (завышена).}

  \vspace{0.3em}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      {\color{navy}\bfseries Что из формулы следует}\\[0.2em]
      {\small Вес n-граммы меняют только сегменты, где она есть: при $x_{ij}
      = 0$ слагаемое — ноль. Вес растёт у класса, которому модель недодаёт
      вероятность, и падает у остальных.}
    \column{0.48\textwidth}
      {\color{brick}\bfseries Первый шаг на нашем корпусе}\\[0.2em]
      {\small Все $p = \tfrac14$. У каждой строки интерфейса с «ить\пр»
      («Удалить…», «Отменить…») ошибка $\tfrac14 - 1 = -\tfrac34$ у
      интерфейса и $+\tfrac14$ у остальных классов. После шага вес «ить\пр»
      для интерфейса — $+0{,}048$, для юридического — $-0{,}019$.}
  \end{columns}
\end{frame}

\begin{frame}{N-граммная модель в формулах}
  {\small
  Вероятность текста раскладывается в цепочку: каждый символ — при условии
  всех предыдущих.
  \формула{P(c_1 c_2 \ldots c_T) = P(c_1)\cdot P(c_2 \mid c_1)\cdot P(c_3 \mid c_1 c_2)
     \cdot\ldots\cdot P(c_T \mid c_1 \ldots c_{T-1})}
  Модель порядка $n$ помнит только $n$ предыдущих символов, а вероятность
  оценивает частотами в обучающем тексте:
  \формула{P(c_t \mid c_{t-n} \ldots c_{t-1}) \;\approx\;
     \frac{\#(c_{t-n} \ldots c_{t-1}\,c_t)}{\#(c_{t-n} \ldots c_{t-1})}}
  $\#(\ldots)$ — сколько раз цепочка встретилась. Обучение здесь — подсчёт,
  градиента нет.}

  \vspace{0.2em}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      {\footnotesize\textbf{Пример, $n = 3$.} В обучающем тексте после «сох»
      всегда шло «р»: «сохранены», «сохраняют», «сохранения». Модель ставит
      «р» с вероятностью \числ{0,921}, ничего не зная о слове.}
    \column{0.48\textwidth}
      {\footnotesize\textbf{Сглаживание.} Цепочка, которой не было, получила
      бы вероятность ноль. Поэтому оценку смешивают с более короткими
      контекстами; на интерактивной странице длинный контекст
      берётся с весом 0,7.}
  \end{columns}
\end{frame}

\begin{frame}{Внимание в формуле: тот же softmax}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      {\small
      У каждого слова $i$ три вектора: запрос $\mathbf q_i$ («что я ищу»),
      ключ $\mathbf k_i$ («что я могу предложить») и значение $\mathbf v_i$
      («что я передам»). С каким весом слово $i$ смотрит на слово $j$:
      \формула{a_{ij} = \mathrm{softmax}_j\!\left(\frac{\mathbf q_i\cdot\mathbf k_j}{\sqrt{d}}\right)}
      Новое представление слова $i$ — смесь значений с этими весами:
      \формула{\mathbf h_i = \sum_{j} a_{ij}\,\mathbf v_j}}
    \column{0.44\textwidth}
      {\small
      Все детали уже встречались. Скалярное произведение измеряет, насколько
      ключ подходит к запросу, — как в косинусе. Softmax превращает баллы в веса, в сумме дающие 1, — как в классификаторе. $\sqrt d$, корень из
      длины векторов, не даёт баллам слишком вырасти.\par
      \vspace{0.4em}
      Сами $\mathbf q, \mathbf k, \mathbf v$ получаются умножением вектора
      слова на матрицы весов. Их и подбирает градиентный спуск.}
  \end{columns}

  \врезка{\textbf{Откуда квадрат.} Для текста из $n$ слов нужно $n \times n$
  скалярных произведений: каждое слово с каждым. Текст вдвое длиннее —
  работы вчетверо больше.}
\end{frame}

\begin{frame}{Почему «совпал — не совпал» здесь не работает}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      {\small
      В классификации ответ либо совпал с правильным, либо нет. Для
      перевода так нельзя: «Сохранить изменения» и «Сохранение изменений»
      допустимы оба, а точное совпадение строк дало бы ноль за верный
      перевод.\par
      \vspace{0.6em}
      Значит, нужна метрика \emph{степени} совпадения с эталоном.}
    \column{0.44\textwidth}
      \врезка{\textbf{Главное ограничение всего подхода:} эталон один, а
      верных переводов много. Всё, что метрика может сказать, — насколько оцениваемый перевод — его называют
      \emph{гипотезой} — похож на один конкретный человеческий вариант.}
  \end{columns}
\end{frame}

\begin{frame}{BLEU: что он на самом деле считает}
  \relax
  {\small Для $n = 1, 2, 3, 4$ берётся доля n-грамм гипотезы, нашедшихся в
  эталоне. Четыре доли усредняются \textbf{геометрически}, результат
  умножается на штраф за слишком короткий ответ. Смысла, грамматичности,
  синонимов и словоформ в этом определении нет.}

  \vspace{0.6em}
  {\small
  \begin{tabular}{@{}Л{0.34}Л{0.34}r@{}}
    \toprule
    эталон & гипотеза & BLEU \\
    \midrule
    Не удалось подключиться к серверу & та же строка & 1,000 \\
    Не удалось подключиться к серверу & \leavevmode\кейс{Удалось} подключиться к серверу & \числ{0{,}779} \\
    Сохранить изменения & Сохранение изменений & 0,408 \\
    \bottomrule
  \end{tabular}}

  \врезка{Противоположный смысл — 0,779, верный перевод другими формами —
  0,408. \textbf{BLEU считает форму, а не смысл.} Л.р. № 3, раздел 3.}
\end{frame}

\begin{frame}{BLEU по формуле}
  \framesubtitle{BLEU одного сегмента в том виде, в каком его считает Л.р. № 3}
  \begin{columns}[T,onlytextwidth]
    \column{0.44\textwidth}
      {\small
      \формула{p_n = \frac{m_n + 1}{t_n + 1}, \qquad n = 1, 2, 3, 4}
      \формула{\mathrm{BP} = \begin{cases} 1, & c > r,\\ e^{\,1 - r/c}, & c \le r \end{cases}}
      \формула{\mathrm{BLEU} = \mathrm{BP}\cdot\sqrt[k]{\,p_1\, p_2 \cdots p_k}}}
    \column{0.52\textwidth}
      {\small
      $t_n$ — сколько n-грамм слов в гипотезе; $m_n$ — сколько из них есть в
      эталоне (каждая — не больше раз, чем там встречается).\par
      \vspace{0.3em}
      $c$ и $r$ — длина гипотезы и эталона в словах. BP (brevity penalty) — штраф только за слишком
      короткий перевод.\par
      \vspace{0.3em}
      $k$ — сколько долей удалось посчитать: в гипотезе из двух слов нет ни
      3-грамм, ни 4-грамм.}
  \end{columns}

  \врезка{\textbf{+1 в числителе и знаменателе — сглаживание.} Без него одна
  нулевая доля обнуляет весь BLEU: произведение с нулём — ноль. Слова
  сравниваются в нижнем регистре, пунктуация отбрасывается.}
\end{frame}

\begin{frame}{chrF по формуле: точность, полнота, F-мера}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}

      важнее.\par}}

      «Сохранить изменения»\par}

        \midrule
        1 & 16 & 16/20 & 16/19 & 0,833 \\
        2 & 14 & 14/19 & 14/18 & 0,769 \\
        3 & 11 & 11/18 & 11/17 & 0,640 \\
        4 &  9 &  9/17 &  9/16 & 0,556 \\
        5 &  7 &  7/16 &  7/15 & 0,461 \\
        6 &  5 &  5/15 &  5/14 & 0,352 \\

\begin{frame}{Согласие разметчиков}
  \relax
  {\small Разметка — это измерение, и у измерения есть надёжность. Если два
  человека размечают один материал по-разному, ненадёжна и разметка, и всё,
  что из неё выведено.}

  \vspace{0.5em}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      {\small
      \textbf{Каппа Коэна} — согласие за вычетом случайного:
      \формула{\kappa = \frac{p_{\text{совп}} - p_{\text{случ}}}{1 - p_{\text{случ}}}}
      Пример Л.р. № 3: совпали 8 ответов из 10, случайно совпало бы 22\,\%:
      $\kappa = \tfrac{0{,}80 - 0{,}22}{1 - 0{,}22} \approx 0{,}744$ —
      «существенное» согласие.}
    \column{0.44\textwidth}
      \врезка{\textbf{Зачем вычитать случайные совпадения.} Если 80\,\% сегментов без
      ошибок, два разметчика, которые вообще не согласны, совпадут в
      \числ{65\,\%} случаев — просто потому, что оба чаще ставят «нет».
      Каппа у них — ноль.}
  \end{columns}

  \vspace{0.5em}
  {\small Расхождения примера — s003: точность или терминология? s006:
  терминология или стиль? Оба ответа можно обосновать, пока руководство для
  разметчиков не решило иначе. Написать его — профессиональная задача
  (Л.р. № 3, задание 8).}
\end{frame}

\begin{frame}{Что связывает все пять блоков}
  \relax
  {\small Машинное обучение не работает с языком — оно работает с числами,
  полученными из языка. Качество этого перехода определяет всё остальное.}

  \vspace{0.7em}
  {\color{navy}\bfseries Три вещи, которые делает лингвист и не может сделать инженер}\\[0.4em]
  \begin{columns}[T,onlytextwidth]
